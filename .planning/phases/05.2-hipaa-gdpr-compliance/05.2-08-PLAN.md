## Objective

Implement GDPR Article 20 Data Portability (right to receive own data in machine-readable format). Background job pattern mirrors the existing reminderWorker pattern exactly.

Design: The export is a background BullMQ job (not synchronous) because a patient with years of appointments, notes, prescriptions, and lab results could have hundreds of documents across 6+ collections. Synchronous response risks timeout at 30-60 seconds.

Export format: JSON with `_metadata` envelope and `data` object containing each collection. Patient's data is decrypted before export (the patient is the data subject receiving their own data — encryption is for at-rest protection, not from the subject themselves).

Purpose: GDPR Article 20 gives patients the right to receive their personal data in a structured, commonly-used, machine-readable format.

Output: `getExportQueue` in reminderQueue.js, new `exportWorker.js`, two new routes in `privacy.js`.

```javascript
'use strict';
/**
 * GDPR Article 20 — Data Export Worker
 *
 * Processes 'gdpr-data-export' BullMQ jobs.
 * Each job: { userId, patientId } where both are ObjectId strings.
 *
 * Steps:
 *   1. Collect all patient data across 6+ collections
 *   2. Data is automatically decrypted by post('init') hooks when documents are loaded
 *   3. Serialize to JSON with _metadata envelope
 *   4. Upload to Cloudinary as private file with 24h signed URL
 *   5. Create in-app Notification with download link
 *   6. Write AuditLog entry
 *
 * PHI note: The export JSON contains plaintext PHI — the patient is the data subject.
 * Transport is HTTPS. The Cloudinary URL expires in 24 hours.
 * The URL is sent to the patient's in-app notification (NOT in API response for security).
 */
const { Worker, Queue } = require('bullmq');
const mongoose = require('mongoose');
const User = require('../models/User');
const Patient = require('../models/Patient');
const Appointment = require('../models/Appointment');
const ConsultationNote = require('../models/ConsultationNote');
const Prescription = require('../models/Prescription');
const LabResult = require('../models/LabResult');
const AuditLog = require('../models/AuditLog');
const { getConnection, getExportQueue } = require('../queues/reminderQueue');
const { uploadBuffer } = require('../utils/cloudinary');

// Safely load optional models
function safeRequireModel(modelName) {
  try { return mongoose.model(modelName); } catch { return null; }
}

async function processExportJob(job) {
  const { userId } = job.data;

  if (!userId) {
    throw new Error('Export job missing userId');
  }

  // 1. Collect User profile (no password hash)
  const user = await User.findById(userId)
    .select('-password -__v')
    .lean();
  if (!user) throw new Error(`User ${userId} not found`);

  // 2. Collect Patient record (post-init hook decrypts fields automatically on lean() — NOTE:
  //    post('init') fires on .exec() but NOT on .lean(). Use non-lean query for encrypted models.)
  const patient = await Patient.findOne({ userId });
  const patientData = patient ? patient.toObject() : null;

  // 3. Collect Appointments
  const appointments = await Appointment.find({ patientId: userId })
    .populate('doctorId', 'name')
    .lean();

  const apptIds = appointments.map(a => a._id);

  // 4. Collect ConsultationNotes — shared only (private notes are doctor's work product)
  const notes = await ConsultationNote.find({
    appointmentId: { $in: apptIds },
    visibility: 'shared',
  });
  const notesData = notes.map(n => n.toObject()); // toObject triggers post-init decrypt

  // 5. Collect Prescriptions
  const prescriptions = await Prescription.find({ patientId: userId });
  const prescriptionsData = prescriptions.map(p => p.toObject());

  // 6. Collect LabResults
  const labResults = await LabResult.find({ patientId: userId });
  const labResultsData = labResults.map(l => l.toObject());

  // 7. Collect Reviews written by patient
  const Review = safeRequireModel('Review');
  const reviewsData = Review
    ? await Review.find({ patientId: userId }).lean()
    : [];

  // 8. Assemble export payload
  const exportPayload = {
    _metadata: {
      exportedAt: new Date().toISOString(),
      version: '1.0',
      dataSubject: {
        userId: user._id,
        name: user.name,
        email: user.email,
        role: user.role,
      },
      generatedBy: 'MediConnect GDPR Export (Article 20)',
    },
    data: {
      profile: {
        ...user,
        password: undefined, // never export password hash — belt and suspenders
      },
      medicalProfile: patientData,
      appointments,
      consultationNotes: notesData,
      prescriptions: prescriptionsData,
      labResults: labResultsData,
      reviews: reviewsData,
    },
  };

  // 9. Upload to Cloudinary as private JSON file with 24h signed URL
  const jsonBuffer = Buffer.from(JSON.stringify(exportPayload, null, 2), 'utf8');
  const uploadResult = await uploadBuffer(
    jsonBuffer,
    `mediconnect/gdpr-exports/${userId}`,
    {
      resource_type: 'raw',
      format: 'json',
      // Cloudinary signed URL expiry is set at download time via signed_url + expires_at
      // File itself has no expiry — we delete it via retentionWorker after 24h
      public_id: `export-${userId}-${Date.now()}`,
    }
  );

  const downloadUrl = uploadResult.secure_url;

  // 10. Create in-app Notification with download URL
  const Notification = safeRequireModel('Notification');
  if (Notification) {
    await Notification.create({
      recipientId: userId,
      type: 'gdpr_export_ready',
      payload: {
        message: 'Your data export is ready. Download link expires in 24 hours.',
        downloadUrl,
        expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000).toISOString(),
      },
    });
  }

  // 11. Write AuditLog
  AuditLog.create({
    actorId:      userId,
    actorRole:    'patient',
    action:       'export',
    resourceType: 'User',
    resourceId:   userId,
    targetUserId: userId,
    outcome:      'success',
    meta:         { exportJobId: job.id, cloudinaryPublicId: uploadResult.public_id },
  }).catch(err => console.error('[audit] export log failed:', err.message));

  return { downloadUrl, jobId: job.id };
}

function startExportWorker() {
  const worker = new Worker('gdpr-data-export', processExportJob, {
    connection: getConnection(),
    concurrency: 2, // exports are heavy; limit concurrency
  });
  worker.on('failed', (job, err) =>
    console.error(`[export] job ${job?.id} failed:`, err.message)
  );
  console.log('[export] worker started');
  return worker;
}

module.exports = { startExportWorker, processExportJob };

/**
 * POST /api/privacy/export
 * GDPR Article 20 — Data Portability
 * Auth: patient only.
 * Enqueues a background export job. Returns 202 with jobId.
 * Patient receives Notification when export is ready (24h download link).
 */
router.post('/export', auth, requireRole('patient'), async (req, res, next) => {
  try {
    const job = await getExportQueue().add('export-patient-data', {
      userId: req.user.id,
    }, {
      jobId: `export-${req.user.id}-${Date.now()}`, // deduplicates concurrent requests
      removeOnComplete: { age: 24 * 60 * 60 }, // keep completed job 24h for status checks
      removeOnFail: { age: 7 * 24 * 60 * 60 }, // keep failed jobs 7 days
    });

    // Fire-and-forget audit log
    AuditLog.create({
      actorId:      req.user.id,
      actorRole:    'patient',
      action:       'export',
      resourceType: 'User',
      resourceId:   req.user.id,
      targetUserId: req.user.id,
      ip:           req.ip,
      userAgent:    req.headers['user-agent'] || null,
      outcome:      'success',
      meta:         { jobId: job.id, status: 'queued' },
    }).catch(err => console.error('[audit] export enqueue log failed:', err.message));

    res.status(202).json({
      message: 'Export job queued. You will receive a notification when your data is ready.',
      jobId: job.id,
    });
  } catch (err) {
    next(err);
  }
});

/**
 * GET /api/privacy/export/:jobId
 * Check status of an export job.
 * Auth: patient only. Patient can only check their own jobs.
 */
router.get('/export/:jobId', auth, requireRole('patient'), async (req, res, next) => {
  try {
    const queue = getExportQueue();
    const job = await queue.getJob(req.params.jobId);

    if (!job) {
      return res.status(404).json({ message: 'Export job not found or expired' });
    }

    // Ownership check: jobId contains userId prefix
    if (!req.params.jobId.includes(req.user.id)) {
      return res.status(403).json({ message: 'Forbidden' });
    }

    const state = await job.getState();
    const result = job.returnvalue;

    res.json({
      jobId: job.id,
      status: state, // 'waiting' | 'active' | 'completed' | 'failed'
      result: state === 'completed' ? { message: 'Check your notifications for download link' } : null,
      failReason: state === 'failed' ? job.failedReason : null,
    });
  } catch (err) {
    next(err);
  }
});
```
