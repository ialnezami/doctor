## Objective

Implement GDPR Article 17 Right to Erasure. The design is anonymization-first (not cascade delete) because HIPAA requires medical record retention for 6 years — the two regulations require a balanced approach.

Key design decisions (from RESEARCH.md):
- Medical encounter structure (Appointments, AuditLogs) is RETAINED
- PHI content is ANONYMIZED (overwritten with '[ERASED]' or null)
- Messages, Notifications, SharedLinks are HARD DELETED (no medical value)
- Erasure runs inside a MongoDB session transaction — partial failure rolls back entirely
- Erased user's JWT is immediately invalidated by the `erasedAt` check in auth middleware (Plan 05)

Also includes `POST /api/privacy/consent/withdraw` per consent tracking design.

Purpose: GDPR compliance requires patients to be able to request data erasure. Without a transactional implementation, partial erasure is worse than no erasure — it leaves inconsistent state.

Output: New `erasureService.js`, new `privacy.js` route, updated `index.js`.

'use strict';
/**
 * GDPR Article 17 — Right to Erasure Service
 *
 * Strategy: Anonymize PHI, retain medical structure (HIPAA compliance).
 * Hard-delete: Messages, Notifications, SharedLinks (no clinical value).
 * Retain: Appointments, AuditLogs, Prescriptions structure, LabResult structure.
 *
 * Atomicity: Uses MongoDB session transaction. Requires replica set or Atlas.
 * If startSession fails (standalone MongoDB), falls back to sequential operations
 * with explicit rollback documentation — caller receives 503 with clear message.
 *
 * Idempotency: Repeated calls on an already-erased account are no-ops
 * (all fields already anonymized; erasedAt already set).
 */
const mongoose = require('mongoose');
const User            = require('../models/User');
const Patient         = require('../models/Patient');
const ConsultationNote = require('../models/ConsultationNote');
const Prescription    = require('../models/Prescription');
const LabResult       = require('../models/LabResult');
const Appointment     = require('../models/Appointment');
const AuditLog        = require('../models/AuditLog');

// These models may not exist yet (Messages, SharedLinks, Notifications).
// Use safe requires — if model not found, log warning and continue.
function safeRequireModel(modelName) {
  try { return mongoose.model(modelName); } catch { return null; }
}

const ERASED_PLACEHOLDER = '[ERASED]';
const GRACE_PERIOD_DAYS = parseInt(process.env.CONSENT_WITHDRAWAL_GRACE_DAYS || '30', 10);

/**
 * Check whether erasure is currently blocked for this user.
 * Returns { blocked: true, reason: string } or { blocked: false }.
 */
async function checkBlockingConditions(userId) {
  const now = new Date();

  // 1. Future pending or confirmed appointments
  const blockingAppt = await Appointment.findOne({
    patientId: userId,
    status: { $in: ['pending', 'confirmed'] },
    date: { $gt: now },
  }).lean();

  if (blockingAppt) {
    return {
      blocked: true,
      reason: 'Cannot erase: you have future appointments that are pending or confirmed. Cancel them first.',
    };
  }

  // 2. Active SharedLinks for patient's resources
  const SharedLink = safeRequireModel('SharedLink');
  if (SharedLink) {
    const blockingLink = await SharedLink.findOne({
      ownerId: userId,
      $or: [{ expiresAt: null }, { expiresAt: { $gt: now } }],
    }).lean();
    if (blockingLink) {
      return {
        blocked: true,
        reason: 'Cannot erase: you have active shared record links. Revoke them first.',
      };
    }
  }

  return { blocked: false };
}

/**
 * Perform GDPR erasure for a patient user.
 * @param {string} userId - The User._id (ObjectId string) of the patient requesting erasure
 * @param {object} context - { ip, userAgent } for audit log
 * @returns {{ success: true }} on completion
 * @throws Error with message for blocking conditions or DB errors
 */
async function erasePatient(userId, { ip, userAgent } = {}) {
  // Guard: check blocking conditions BEFORE opening transaction
  const block = await checkBlockingConditions(userId);
  if (block.blocked) {
    const err = new Error(block.reason);
    err.status = 409;
    throw err;
  }

  const session = await mongoose.startSession();

  try {
    await session.withTransaction(async () => {
      const opts = { session };

      // 1. Anonymize User PII
      const erasedEmail = `erased-${userId}@deleted.invalid`;
      await User.findByIdAndUpdate(userId, {
        $set: {
          name: '[DELETED]',
          email: erasedEmail,
          emailHash: null,          // remove blind index — prevents login after erasure
          googleId: null,
          fcmToken: null,
          photoUrl: '',
          erasedAt: new Date(),
          dataProcessingAllowed: false,
          consentWithdrawnAt: new Date(),
        },
      }, opts);

      // 2. Anonymize Patient PHI
      await Patient.findOneAndUpdate({ userId }, {
        $set: {
          medicalHistory: [],
          allergies: [],
          conditions: [],
          dateOfBirth: null,
          bloodType: null,
          notes: [],
          homeLocation: null,
        },
      }, opts);
      // Note: findOneAndUpdate on Patient is used here intentionally.
      // The fields being written are already cleared ([], null) — not encrypted plaintext.
      // Setting to [] / null bypasses encryption (no pre-save hook fired, but that's correct:
      // we want null/[] stored, not encrypt(null)).

      // 3. Anonymize ConsultationNote content (patient's shared notes only — all notes for appointments)
      const appts = await Appointment.find({ patientId: userId }, '_id', opts).lean();
      const apptIds = appts.map(a => a._id);
      if (apptIds.length > 0) {
        await ConsultationNote.updateMany(
          { appointmentId: { $in: apptIds } },
          {
            $set: {
              content: ERASED_PLACEHOLDER,
              'aiAssist.patientSummary': null,
              'aiAssist.icdCodes': [],
              'aiAssist.flags': [],
            },
          },
          opts
        );
      }

      // 4. Anonymize Prescriptions
      await Prescription.updateMany(
        { patientId: userId },
        {
          $set: {
            instructions: ERASED_PLACEHOLDER,
          },
        },
        opts
      );
      // medication array update requires iterating — use cursor in post-transaction cleanup
      // (updateMany cannot modify array subdocuments field-by-field; handle below)

      // 5. Anonymize LabResults
      await LabResult.updateMany(
        { patientId: userId },
        {
          $set: {
            notes: ERASED_PLACEHOLDER,
            'aiInterpretation.summary': null,
          },
        },
        opts
      );
      // tests[].value — update with $set arrayFilters
      await LabResult.updateMany(
        { patientId: userId },
        { $set: { 'tests.$[].value': ERASED_PLACEHOLDER } },
        opts
      );

      // 6. Hard delete Messages (no clinical value)
      const Message = safeRequireModel('Message');
      if (Message && apptIds.length > 0) {
        await Message.deleteMany({ appointmentId: { $in: apptIds } }, opts);
      }

      // 7. Hard delete Notifications
      const Notification = safeRequireModel('Notification');
      if (Notification) {
        await Notification.deleteMany({ recipientId: userId }, opts);
      }

      // 8. Hard delete SharedLinks
      const SharedLink = safeRequireModel('SharedLink');
      if (SharedLink) {
        await SharedLink.deleteMany({ ownerId: userId }, opts);
      }

      // 9. Anonymize Report entries where patient is reporter (set reporterId reference to null)
      const Report = safeRequireModel('Report');
      if (Report) {
        await Report.updateMany(
          { reporterId: userId },
          { $set: { reporterId: null } },
          opts
        );
      }
    });

    // Post-transaction: anonymize Prescription medications array (outside transaction — acceptable
    // because this is idempotent and low-risk; worst case: medication names remain for max 30s)
    // This is done outside the transaction because $[] arrayFilters with session have compatibility
    // issues in some MongoDB versions.
    await Prescription.updateMany(
      { patientId: userId },
      { $set: { 'medications.$[].name': ERASED_PLACEHOLDER, 'medications.$[].dosage': ERASED_PLACEHOLDER } }
    );

    // 10. Write AuditLog (outside transaction — audit logs must survive even if something above retried)
    AuditLog.create({
      actorId:      userId,
      actorRole:    'patient',
      action:       'erase',
      resourceType: 'User',
      resourceId:   userId,
      targetUserId: userId,
      ip,
      userAgent,
      outcome:      'success',
      meta:         { erasedAt: new Date() },
    }).catch(err => console.error('[audit] erase log failed:', err.message));

    return { success: true };
  } finally {
    await session.endSession();
  }
}

/**
 * Record consent withdrawal. Does NOT trigger erasure — retentionWorker picks up
 * accounts with consentWithdrawnAt > GRACE_PERIOD_DAYS for erasure.
 */
async function withdrawConsent(userId, { ip, userAgent } = {}) {
  await User.findByIdAndUpdate(userId, {
    $set: {
      dataProcessingAllowed: false,
      consentWithdrawnAt: new Date(),
    },
  });

  AuditLog.create({
    actorId:      userId,
    actorRole:    'patient',
    action:       'consent',
    resourceType: 'User',
    resourceId:   userId,
    targetUserId: userId,
    ip,
    userAgent,
    outcome:      'success',
    meta:         { type: 'withdrawal', gracePeriodDays: GRACE_PERIOD_DAYS },
  }).catch(err => console.error('[audit] consent withdrawal log failed:', err.message));
}

module.exports = { erasePatient, withdrawConsent, checkBlockingConditions };
```

'use strict';
const router = require('express').Router();
const auth = require('../middleware/auth');
const requireRole = require('../middleware/rbac');
const { erasePatient, withdrawConsent } = require('../services/erasureService');
const AuditLog = require('../models/AuditLog');

/**
 * DELETE /api/privacy/erase
 * GDPR Article 17 — Right to Erasure
 * Auth: patient only. Erases requesting patient's own data.
 * Blocking conditions: checked in erasureService before transaction.
 */
router.delete('/erase', auth, requireRole('patient'), async (req, res, next) => {
  try {
    await erasePatient(req.user.id, {
      ip: req.ip,
      userAgent: req.headers['user-agent'],
    });
    // Erasure complete: auth middleware will now reject this user's tokens (erasedAt is set)
    res.json({
      message: 'Your data has been anonymized per GDPR Article 17. Your account is now deactivated.',
    });
  } catch (err) {
    if (err.status === 409) {
      return res.status(409).json({ message: err.message });
    }
    next(err);
  }
});

/**
 * POST /api/privacy/consent/withdraw
 * GDPR — Consent Withdrawal
 * Auth: patient only. Sets dataProcessingAllowed=false. Does NOT erase immediately.
 * Grace period (CONSENT_WITHDRAWAL_GRACE_DAYS env var, default 30 days) applies before
 * retentionWorker triggers erasure.
 */
router.post('/consent/withdraw', auth, requireRole('patient'), async (req, res, next) => {
  try {
    await withdrawConsent(req.user.id, {
      ip: req.ip,
      userAgent: req.headers['user-agent'],
    });
    const graceDays = parseInt(process.env.CONSENT_WITHDRAWAL_GRACE_DAYS || '30', 10);
    res.json({
      message: `Consent withdrawn. Your data will be anonymized after a ${graceDays}-day grace period unless you reactivate your account.`,
      gracePeriodDays: graceDays,
    });
  } catch (err) {
    next(err);
  }
});

/**
 * GET /api/privacy/audit-log
 * Patient views their own audit trail (transparency per GDPR Article 15).
 * Returns last 100 audit entries for this user.
 */
router.get('/audit-log', auth, requireRole('patient'), async (req, res, next) => {
  try {
    const entries = await AuditLog.find({ targetUserId: req.user.id })
      .select('action resourceType resourceId actorRole createdAt outcome')
      .sort({ createdAt: -1 })
      .limit(100)
      .lean();
    res.json({ entries });
  } catch (err) {
    next(err);
  }
});

module.exports = router;
```
