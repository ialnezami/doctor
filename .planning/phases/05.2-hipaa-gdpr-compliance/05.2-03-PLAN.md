## Objective

Apply AES-256-GCM encryption hooks to all four PHI-bearing Mongoose models. After this plan:
- PHI fields are encrypted before MongoDB writes (pre-save hook)
- Decrypted transparently on read (post-init hook)
- Legacy plaintext records are returned as-is (fieldEncryption.decrypt handles migration sentinel)
- No route code changes are needed — the hooks are transparent

This plan does NOT touch routes. Routes that use `findOneAndUpdate` on these models must be refactored in Plan 04 — that refactor depends on these hooks existing.

Purpose: The `decrypt` function's `|` sentinel means existing records remain readable after deployment with zero downtime.

Output: 4 updated model files with Mongoose pre/post hooks.

<read_first>

<action>
Rewrite `apps/api/src/models/Patient.js` to add pre-save encryption and post-init decryption hooks.

PHI fields to encrypt:
- `medicalHistory` — Array of strings
- `allergies` — Array of strings
- `conditions` — Array of strings
- `dateOfBirth` — Date (convert to ISO string for encryption, restore as Date on decrypt)
- `bloodType` — String (enum — encrypt the value; the enum validator runs BEFORE pre-save, so validation sees plaintext; after pre-save, MongoDB stores ciphertext)

Important: `notes[].content` inside Patient.notes is doctor clinical notes on the patient profile (distinct from ConsultationNote). Encrypt `notes[].content` as well.

Fields NOT encrypted (must remain queryable): `userId`, `homeLocation`, `city`, `photoUrl`, `createdAt`, `updatedAt`.

Add after the existing index declaration:

```javascript
const { encrypt, decrypt } = require('../utils/fieldEncryption');

// Scalar PHI fields (String or Date stored as string)
const SCALAR_PHI = ['bloodType'];
// Array PHI fields (each element is a string)
const ARRAY_PHI = ['medicalHistory', 'allergies', 'conditions'];

patientSchema.pre('save', function (next) {
  try {
    SCALAR_PHI.forEach(field => {
      if (this.isModified(field) && this[field] != null) {
        this[field] = encrypt(String(this[field]));
      }
    });

    ARRAY_PHI.forEach(field => {
      if (this.isModified(field) && Array.isArray(this[field])) {
        this[field] = this[field].map(v => (v != null ? encrypt(String(v)) : v));
      }
    });

    // dateOfBirth: encrypt as ISO string; stored as mixed (not Date) after encryption
    if (this.isModified('dateOfBirth') && this.dateOfBirth != null) {
      this.dateOfBirth = encrypt(new Date(this.dateOfBirth).toISOString());
    }

    // notes[].content: encrypt each doctor note on patient profile
    if (this.isModified('notes') && Array.isArray(this.notes)) {
      this.notes = this.notes.map(note => {
        if (note.content && !String(note.content).includes('|')) {
          return { ...note.toObject ? note.toObject() : note, content: encrypt(note.content) };
        }
        return note;
      });
    }

    next();
  } catch (err) {
    next(err);
  }
});

patientSchema.post('init', function (doc) {
  try {
    SCALAR_PHI.forEach(field => {
      if (doc[field] != null) doc[field] = decrypt(doc[field]);
    });

    ARRAY_PHI.forEach(field => {
      if (Array.isArray(doc[field])) {
        doc[field] = doc[field].map(v => (v != null ? decrypt(String(v)) : v));
      }
    });

    if (doc.dateOfBirth != null) {
      const decrypted = decrypt(String(doc.dateOfBirth));
      doc.dateOfBirth = decrypted ? new Date(decrypted) : null;
    }

    if (Array.isArray(doc.notes)) {
      doc.notes = doc.notes.map(note => {
        if (note.content) note.content = decrypt(note.content);
        return note;
      });
    }
  } catch (err) {
    // Decrypt failure on read should not crash the application
    // Log without PHI content — only the document ID
    console.error('[encrypt] Patient decrypt error on doc', doc._id, ':', err.message);
  }
});
```

Critical: The `isModified(field)` guard in pre-save prevents double-encryption when a document is loaded and re-saved without touching the PHI field. Without this guard, an already-encrypted ciphertext would be encrypted again, corrupting data permanently.
</action>

**Prescription — rewrite `apps/api/src/models/Prescription.js`:**

PHI fields: `instructions` (string). For `medications`, encrypt only `name` field per medication subdocument (dosage, frequency, duration are clinical metadata but still PHI — encrypt them all for safety).

Note: `medications` is an array of subdocuments (medicationSchema). Subdoc fields must be encrypted/decrypted by iterating the array.

```javascript
const { encrypt, decrypt } = require('../utils/fieldEncryption');

// Medication subdoc fields to encrypt
const MED_FIELDS = ['name', 'dosage', 'frequency', 'duration'];

prescriptionSchema.pre('save', function (next) {
  try {
    if (this.isModified('instructions') && this.instructions != null) {
      this.instructions = encrypt(this.instructions);
    }
    if (this.isModified('medications') && Array.isArray(this.medications)) {
      this.medications = this.medications.map(med => {
        const m = med.toObject ? med.toObject() : { ...med };
        MED_FIELDS.forEach(f => {
          if (m[f] != null && !String(m[f]).includes('|')) {
            m[f] = encrypt(String(m[f]));
          }
        });
        return m;
      });
    }
    next();
  } catch (err) { next(err); }
});

prescriptionSchema.post('init', function (doc) {
  try {
    if (doc.instructions != null) doc.instructions = decrypt(doc.instructions);
    if (Array.isArray(doc.medications)) {
      doc.medications = doc.medications.map(med => {
        MED_FIELDS.forEach(f => {
          if (med[f] != null) med[f] = decrypt(String(med[f]));
        });
        return med;
      });
    }
  } catch (err) {
    console.error('[encrypt] Prescription decrypt error on doc', doc._id, ':', err.message);
  }
});
```
