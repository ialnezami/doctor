    'use strict';
    const NodeCache = require('node-cache');
    const MAX_TURNS = 20;
    const TTL_SECONDS = 1800;
    const cache = new NodeCache({ stdTTL: TTL_SECONDS, checkperiod: 300, useClones: false });

    function getHistory(userId) { return cache.get(userId) || []; }

    function snapshotHistory(userId) {
      // Deep copy — mutations to snapshot do NOT affect stored history
      return JSON.parse(JSON.stringify(cache.get(userId) || []));
    }

    function appendAndSave(userId, userMessage, assistantMessage) {
      let history = cache.get(userId) || [];
      history = [...history, { role: 'user', content: userMessage }, { role: 'assistant', content: assistantMessage }];
      if (history.length > MAX_TURNS) history = history.slice(history.length - MAX_TURNS);
      cache.set(userId, history);
    }

    function clearSession(userId) { cache.del(userId); }

    module.exports = { getHistory, snapshotHistory, appendAndSave, clearSession, MAX_TURNS, TTL_SECONDS };

    const TRIAGE_RE = /<triage>([\s\S]*?)<\/triage>/;

    function parseTriage(text) {
      if (typeof text !== 'string') return null;
      const match = text.match(TRIAGE_RE);
      if (!match) return null;
      let parsed;
      try { parsed = JSON.parse(match[1].trim()); } catch { return null; }

      const specialties = Array.isArray(parsed.specialties) ? parsed.specialties.filter(s => typeof s === 'string').slice(0, 5) : [];

        summary: typeof parsed.summary === 'string' ? parsed.summary.slice(0, 500) : '',

    **chatbotService.js** — export `TRIAGE_SYSTEM_PROMPT` (verbatim from RESEARCH Pattern 3) and `streamChatResponse(res, history, systemPrompt, { requestId })`.

    **Stream lifecycle contract — the service does NOT own res.end():**
    - Set SSE headers first: `Content-Type: text/event-stream`, `Cache-Control: no-cache`, `Connection: keep-alive`, `X-Accel-Buffering: no`
    - Guard: `if (!process.env.ANTHROPIC_API_KEY) { res.status(503).json({ message: 'AI service unavailable' }); return null; }`
    - Instantiate `new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY })`
    - AbortController with `setTimeout(() => controller.abort(), 25000)`
    - `for await (const event of stream)` — accumulate text on `content_block_delta`+`text_delta`, write `data: {"type":"delta","text":"..."}` SSE events
    - Error handling (catch, NOT finally):
      - On AbortError: `res.write('data: ' + JSON.stringify({ type: 'error', message: 'Response timed out' }) + '\n\n')` then re-throw
      - On any other error: `res.write('data: ' + JSON.stringify({ type: 'error', message: 'AI service temporarily unavailable' }) + '\n\n')` then re-throw
    - Finally block: `clearTimeout(timeoutId)` ONLY — do NOT call `res.write('[DONE]')` or `res.end()` here
    - Return `accumulatedText` (the full assembled response string) so the route can parse triage and close the stream
    - Log ONLY: requestId, userId (from caller), tokenCount estimate, durationMs — NEVER history or accumulated text content

    **Why this matters:** The route handler calls `streamChatResponse` and awaits it. After it resolves, the route parses triage, queries doctors, then writes the structured `done` SSE event (with urgency + doctors) followed by `[DONE]` and calls `res.end()`. If the service calls `res.end()` first, the stream is already closed and the `done` event (triage + doctors) is silently dropped — clients never receive it.

    **routes/chatbot.js** — implement 3 routes in this order (middleware chain matters):
    ```javascript
    'use strict';
    const router = require('express').Router();
    const crypto = require('crypto');
    const { body, query, validationResult } = require('express-validator');
    const auth = require('../middleware/auth');
    const requireRole = require('../middleware/rbac');
    const { chatbotLimiter } = require('../middleware/rateLimiter');
    const { streamChatResponse, TRIAGE_SYSTEM_PROMPT } = require('../services/chatbotService');
    const { snapshotHistory, appendAndSave, clearSession } = require('../utils/sessionStore');
    const { parseTriage } = require('../utils/triageParser');
    const { getRankedDoctors } = require('../utils/doctorRanking');

    const validate = (req, res, next) => {
      const errors = validationResult(req);
      if (!errors.isEmpty()) return res.status(422).json({ errors: errors.array() });
      next();
    };

    // POST /api/chatbot/message
    router.post('/message',
      auth,
      requireRole('patient'),
      chatbotLimiter,
      body('message').isString().trim().isLength({ min: 1, max: 2000 }),
      body('lat').optional().isFloat({ min: -90, max: 90 }),
      body('lng').optional().isFloat({ min: -180, max: 180 }),
      validate,
      async (req, res) => {
        const requestId = crypto.randomUUID();
        const userId = req.user.id;
        const { message, lat, lng } = req.body;

        // Pitfall 6: snapshot BEFORE stream to survive session reset race
        const historySnapshot = snapshotHistory(userId);
        const historyForClaude = [...historySnapshot, { role: 'user', content: message }];

        const startedAt = Date.now();
        let accumulated;
        try {
          accumulated = await streamChatResponse(res, historyForClaude, TRIAGE_SYSTEM_PROMPT, { requestId });
        } catch (err) {
          // Service already wrote SSE error event and re-threw — close the stream
          console.error(`[chatbot] requestId=${requestId} userId=${userId} error=${err.message}`);
          res.write('data: [DONE]\n\n');
          res.end();
          return;
        }

        if (accumulated === null) return; // 503 already sent (no SSE headers set)

        // Parse triage block AFTER stream done, before doctors query
        const triage = parseTriage(accumulated);
        let doctors = [];
        let specialtyFallback = false;
        const isEmergency = triage?.urgency === 'emergency';

        if (triage && !isEmergency && lat != null && lng != null && triage.ready_for_referral) {
          try {
            const specialty = triage.specialties?.[0];
            const ranked = await getRankedDoctors({ specialty, lat, lng, limit: 5 });
            doctors = ranked.doctors;
            specialtyFallback = ranked.specialtyFallback;
          } catch (err) {
            console.error(`[chatbot] requestId=${requestId} doctorRanking error=${err.message}`);
          }
        }

        // Persist history (Pitfall 6: uses snapshot, safe against mid-stream reset)
        appendAndSave(userId, message, accumulated);

        // Route owns stream lifecycle: write done event then close
        res.write(`data: ${JSON.stringify({ type: 'done', urgency: triage?.urgency || null, specialties: triage?.specialties || [], summary: triage?.summary || null, emergency: isEmergency, doctors, specialtyFallback })}\n\n`);
        res.write('data: [DONE]\n\n');
        res.end();

        console.log(`[chatbot] requestId=${requestId} userId=${userId} urgency=${triage?.urgency || 'unparsed'} durationMs=${Date.now() - startedAt} doctorsReturned=${doctors.length}`);
      }
    );

    // GET /api/chatbot/doctors — direct ranked doctor lookup (used by UI if user overrides specialty)
    router.get('/doctors',
      auth,
      requireRole('patient'),
      chatbotLimiter,
      query('specialty').optional().isString().isLength({ max: 100 }),
      query('lat').isFloat({ min: -90, max: 90 }),
      query('lng').isFloat({ min: -180, max: 180 }),
      query('limit').optional().isInt({ min: 1, max: 20 }),
      validate,
      async (req, res, next) => {
        try {
          const { specialty, lat, lng, limit } = req.query;
          const result = await getRankedDoctors({ specialty, lat, lng, limit: parseInt(limit) || 5 });
          res.json(result);
        } catch (err) { next(err); }
      }
    );

    // DELETE /api/chatbot/session — clear caller's conversation
    router.delete('/session', auth, requireRole('patient'), (req, res) => {
      clearSession(req.user.id);
      res.status(204).end();
    });

    module.exports = router;
    ```
